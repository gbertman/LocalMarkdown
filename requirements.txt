# LocalMarkdown MCP server — Python dependencies
# Tested with Python 3.10–3.12.  Install with:  pip install -r requirements.txt
#
# Docling pulls in PyTorch (for its layout + table models) and handles PDF, Word,
# Excel, PowerPoint and HTML in one pipeline, so you do NOT need PyMuPDF/openpyxl/
# python-docx separately. NOTE: recent Docling versions no longer bundle an OCR
# engine, so EasyOCR is listed explicitly below for image/scanned-PDF OCR.

# --- MCP server framework ---------------------------------------------------
mcp[cli]>=1.2.0

# --- Directory monitoring ---------------------------------------------------
watchdog>=4.0.0

# --- Document + image conversion (PDF/DOCX/XLSX/PPTX/HTML/images, OCR) -------
docling>=2.0.0
# Docling no longer bundles an OCR engine in recent versions; this project's
# pipeline requests EasyOCR explicitly, so install it alongside Docling.
# (EasyOCR reuses the torch installed above; on first use it downloads ~64 MB
# of detection/recognition models per OCR language.)
easyocr>=1.7.0

# --- Media transcription (audio + video) ------------------------------------
# CTranslate2 backend — much faster and lighter than openai-whisper, no torch.
# PyAV (bundled) decodes audio/video; the `ffmpeg` binary is recommended for
# the widest container/codec coverage (apt install ffmpeg / winget Gyan.FFmpeg).
faster-whisper>=1.0.0

# --- Image description (the "vision" piece) ---------------------------------
Pillow>=10.0.0          # technical image metadata (size/mode/format)

# Pick a describer backend via LM_IMAGE_DESCRIBE (default: auto). The local-LLM
# options below cost ZERO API tokens and need NO extra Python package — they
# talk to a local server over HTTP using only the standard library.
#
#   * LM_IMAGE_DESCRIBE=ollama   (recommended local LLM — no tokens)
#       Install Ollama from https://ollama.com, then pull a vision model:
#         ollama pull llama3.2-vision       # or llava / moondream / qwen2.5-vl
#       Configure with LM_OLLAMA_MODEL / LM_OLLAMA_HOST.
#
#   * LM_IMAGE_DESCRIBE=openai   (any OpenAI-compatible local server)
#       LM Studio / llama.cpp --server / vLLM. Set LM_OPENAI_BASE_URL + LM_OPENAI_MODEL.
#
#   * LM_IMAGE_DESCRIBE=anthropic   (cloud Claude vision — uses API tokens)
#       Uncomment the SDK below and set ANTHROPIC_API_KEY.
# anthropic>=0.40.0
#
#   * LM_IMAGE_DESCRIBE=blip   (local caption model, no LLM, ~1 GB)
# transformers>=4.40.0
#
# auto (default): uses a local Ollama server if one is reachable, else Claude if
# ANTHROPIC_API_KEY is set, else local BLIP, else skips descriptions.

# --- Email parsing ----------------------------------------------------------
# .eml and .mbox are handled entirely by the Python standard library (no
# dependency). .mbox mailboxes (Unix / Google Takeout) are burst into one
# Markdown per message via the stdlib `mailbox` module.
# .msg (Outlook single message) needs extract-msg:
extract-msg>=0.45.0
# .pst (Outlook mailbox) is exploded into messages by the `readpst` binary from
# pst-utils -- a SYSTEM package, NOT pip:   sudo apt install pst-utils
# Email attachments are written as their own Markdown files, each tagged in its
# frontmatter (attachment_of / email_subject / container) so you can trace which
# email they came from.
